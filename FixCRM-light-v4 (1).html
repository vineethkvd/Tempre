using Business.Contracts; using DataAccess.Dto.Request; using Microsoft.Extensions.Configuration; using Microsoft.Extensions.Hosting; using Microsoft.Extensions.Logging; using System; using System.IO; using System.Threading; using System.Threading.Tasks; namespace ICICI_DC_Scheduler.Schedulers { public class IciciDcWorker( ILogger logger, IServiceWrapper service, IConfiguration configuration) : BackgroundService { private static readonly object _logLock = new object(); protected override async Task ExecuteAsync(CancellationToken stoppingToken) { int intervalMinutes = configuration.GetValue("Schedule:IntervalMinutes", 60); if (intervalMinutes <= 0) { intervalMinutes = 60; } LogInfo($"IciciDcWorker started. Polling every {intervalMinutes} minutes."); await ProcessSchedulerAsync(); using var timer = new PeriodicTimer(TimeSpan.FromMinutes(intervalMinutes)); try { while (await timer.WaitForNextTickAsync(stoppingToken)) { await ProcessSchedulerAsync(); } } catch (OperationCanceledException) when (stoppingToken.IsCancellationRequested) { LogInfo("IciciDcWorker stopped."); } } private async Task ProcessSchedulerAsync() { try { LogInfo($"Worker running at: {DateTimeOffset.Now:yyyy-MM-dd HH:mm:ss zzz}"); var postReqDto = new PostReqDto { p_flag = "ICICIDC", p_pagevalue = "1", p_paravalue = "1" }; var data = await service.icici_service.ICICI_DC_SchedulerPostService(postReqDto); if (data == null) { LogWarning("ICICI_DC_SchedulerPostService returned null."); } else { LogInfo("ICICI_DC_SchedulerPostService executed successfully."); } } catch (Exception ex) { LogError(ex, "Error in IciciDcWorker"); } } private void WriteToFile(string message) { string path = Path.Combine(AppContext.BaseDirectory, "Logs"); string dateStr = DateTime.Now.ToString("yyyy_MM_dd"); string filepath = Path.Combine(path, $"ServiceLog_{dateStr}.txt"); string formattedMessage = $"[{DateTime.Now:yyyy-MM-dd HH:mm:ss}] {message}"; lock (_logLock) { if (!Directory.Exists(path)) { Directory.CreateDirectory(path); } using StreamWriter sw = new StreamWriter(filepath, append: true); sw.WriteLine(formattedMessage); } } private void LogInfo(string message) { logger.LogInformation("{Message}", message); WriteToFile(message); } private void LogWarning(string message) { logger.LogWarning("{Message}", message); WriteToFile($"WARNING: {message}"); } private void LogError(Exception ex, string message) { logger.LogError(ex, "{Message}", message); WriteToFile($"ERROR: {message} | Exception: {ex.Message} | StackTrace: {ex.StackTrace}"); } } }
