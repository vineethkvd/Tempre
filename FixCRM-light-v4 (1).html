using System; using System.IO; using System.Linq; namespace Business.Helpers { public class LogCleanupService { private readonly string _logDirectory; private readonly int _retentionMonths; private static readonly string[] LogExtensions = { ".log", ".txt" }; public LogCleanupService(string logDirectory, int retentionMonths = 7) { _logDirectory = logDirectory; _retentionMonths = retentionMonths > 0 ? retentionMonths : 7; } public void CleanupOldLogs() { if (!Directory.Exists(_logDirectory)) { Directory.CreateDirectory(_logDirectory); return; } // Delete any log file older than 7 months DateTime cutoffDate = DateTime.Now.AddMonths(-_retentionMonths); var logFiles = Directory.EnumerateFiles(_logDirectory, "*.*", SearchOption.AllDirectories) .Where(file => LogExtensions.Contains(Path.GetExtension(file), StringComparer.OrdinalIgnoreCase)); foreach (var logFile in logFiles) { try { DateTime lastWriteTime = File.GetLastWriteTime(logFile); if (lastWriteTime < cutoffDate) { File.Delete(logFile); } } catch (Exception ex) { Console.WriteLine($"Error deleting file {logFile}: {ex.Message}"); } } } } }
